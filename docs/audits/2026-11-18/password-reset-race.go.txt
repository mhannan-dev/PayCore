package main

import (
	"context"
	"crypto/sha256"
	"fmt"
	"log"
	"strings"
	"time"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
	"paycore/api/internal/config"
	"paycore/api/internal/database"
	"paycore/api/internal/model"
	"paycore/api/internal/repository"
	"paycore/api/internal/service"
	"paycore/api/migrations"
)

type captchaOK struct{}

func (captchaOK) Verify(string, string) bool { return true }

type resetBeforeSession struct {
	*repository.UserRepository
	reset       *repository.PasswordResetRepository
	resetDigest []byte
	newHash     string
}

func (r *resetBeforeSession) CreateSession(ctx context.Context, id string, digest []byte, expiry time.Time) error {
	if _, err := r.reset.ResetPassword(ctx, r.resetDigest, r.newHash); err != nil {
		return err
	}
	return r.UserRepository.CreateSession(ctx, id, digest, expiry)
}

func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}
func run() error {
	ctx, cancel := context.WithTimeout(context.Background(), 40*time.Second)
	defer cancel()
	admin, err := database.Open(ctx, config.Load().DatabaseURL)
	if err != nil {
		return err
	}
	defer admin.Close()
	name := fmt.Sprintf("auth_audit_%d", time.Now().UnixNano())
	schema := pgx.Identifier{name}.Sanitize()
	if _, err = admin.Exec(ctx, "CREATE SCHEMA "+schema); err != nil {
		return err
	}
	defer func() {
		cleanup, stop := context.WithTimeout(context.Background(), 10*time.Second)
		defer stop()
		if _, err := admin.Exec(cleanup, "DROP SCHEMA "+schema+" CASCADE"); err != nil {
			log.Print("Audit schema cleanup failed")
		}
	}()
	cfg, err := pgxpool.ParseConfig(config.Load().DatabaseURL)
	if err != nil {
		return err
	}
	cfg.ConnConfig.RuntimeParams["search_path"] = name
	pool, err := pgxpool.NewWithConfig(ctx, cfg)
	if err != nil {
		return err
	}
	defer pool.Close()
	if err = migrations.Apply(ctx, pool); err != nil {
		return err
	}
	users := repository.NewUserRepository(pool)
	oldHash, err := service.HashPassword("OldPwd123!")
	if err != nil {
		return err
	}
	newHash, err := service.HashPassword("NewPwd123!")
	if err != nil {
		return err
	}
	u, err := users.Create(ctx, "Audit", nil, "audit@example.test", oldHash, []string{"OPERATOR"})
	if err != nil {
		return err
	}
	digest := sha256.Sum256([]byte(strings.Repeat("a", 64)))
	resets := repository.NewPasswordResetRepository(pool)
	if _, err = resets.IssueResetToken(ctx, u.Email, digest[:], time.Now().Add(time.Minute), time.Second); err != nil {
		return err
	}
	store := &resetBeforeSession{UserRepository: users, reset: resets, resetDigest: digest[:], newHash: newHash}
	auth, err := service.NewAuthService(store, captchaOK{})
	if err != nil {
		return err
	}
	login, err := auth.Login(ctx, model.LoginRequest{Email: u.Email, Password: "OldPwd123!"})
	if err != nil {
		return err
	}
	_, err = auth.CurrentUser(ctx, login.Token)
	fmt.Printf("credential_reset_race_reproduced=%t\n", err == nil)
	return err
}
